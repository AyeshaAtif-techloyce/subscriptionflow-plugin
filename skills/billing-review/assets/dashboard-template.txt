    const fail = (DATA.failed_payments||[]).find(f => f.invoice === x.number);
    box.innerHTML = `<div class="detail"><span class="label">${esc(x.number)} · ${esc(x.customer)}</span>
      <div class="kv"><span>Status</span><span>${esc(x.status)}</span><span>Invoice total</span><span>${money(x.total)}</span><span>Amount due</span><span>${money(x.amount_due)}</span>
      <span>Invoice date</span><span>${fmtDate(x.invoice_date)}</span><span>Due date</span><span>${fmtDate(x.due_date)}</span>
      <span>Saved payment method</span><span>${x.has_payment_method ? "Yes" : "No"}</span>${fail ? `<span>Last charge</span><span>Failed ${fmtDate(fail.date)} (${esc(fail.reason.replace(/_/g," "))})</span>` : ""}</div>
      ${x.action ? `<div>To ${x.action.type==="send_invoice"?"email this invoice to the customer":"look into this invoice"}, say this in the chat. Claude shows a preview first and changes nothing until you confirm.</div>
      <input class="say" readonly value="${esc(x.action.say)}" id="sf-say" aria-label="Sentence to send"><div><button class="btn" id="sf-copy">Copy sentence</button> <button class="btn ghost" id="sf-close">Close</button></div>`
      : `<div>No action needed yet.</div><div><button class="btn ghost" id="sf-close">Close</button></div>`}</div>`;
    $("#sf-close").onclick = () => { sel = null; render(); };
    if (x.action) $("#sf-copy").onclick = () => copySay(x.action.say);
  }
  function toast(t){ const el = $("#sf-toast"); el.textContent = t; el.hidden = false; clearTimeout(toast.t); toast.t = setTimeout(() => el.hidden = true, 4200); }
  function copySay(s){
    const done = () => toast(`Copied "${s}". Paste it in the chat; Claude will preview before anything changes.`);
    try { navigator.clipboard.writeText(s).then(done, () => { fallback(s); }); } catch(e){ fallback(s); }
    function fallback(s){ const i = document.createElement("input"); i.value = s; document.body.appendChild(i); i.select(); try { document.execCommand("copy"); done(); } catch(_) { toast(`Say: "${s}"`); } i.remove(); }
  }

  // Side lists
  const fp = DATA.failed_payments || [];
  $("#sf-failed").innerHTML = fp.length ? fp.map(f => `<div class="li"><strong>${esc(f.customer)}</strong><span>${money(f.amount)}</span><span class="sub">${fmtDate(f.date)} · ${esc((f.reason||"").replace(/_/g," "))}${f.invoice?` · ${esc(f.invoice)}`:""}</span></div>`).join("") : `<div class="empty">No failed payments.</div>`;
  const rn = (DATA.renewals || []).slice().sort((a,b) => a.renews_on < b.renews_on ? -1 : 1);
  $("#sf-renew").innerHTML = rn.length ? rn.map(r => `<div class="li"><strong>${esc(r.customer)}</strong><span>${money(r.amount)}</span><span class="sub">${esc(r.plan)} · renews ${fmtDate(r.renews_on)}</span></div>`).join("") : `<div class="empty">No renewals in the next 30 days.</div>`;
  $("#sf-notes").innerHTML = (DATA.notes||[]).map(n => `<li>${esc(n)}</li>`).join("");

  function render(){ renderAging(); renderTable(); renderDetail(); }
  render();
})();
