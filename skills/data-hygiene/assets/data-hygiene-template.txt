<script>
const SAMPLE = {
  generated_at:"2026-10-20", account:"Sample account",
  checks:[
    {id:"stale_due",title:"Invoices past due still marked \"Due\"",severity:"high",status:"found",count:12,detail:"12 open invoices have a due date in the past but their status was never moved to Overdue. The oldest is 8 months late.",why:"Overdue reminders and dunning key off the status, so these are never chased.",fix:"Check the job that moves invoices to Overdue, then update the status.",examples:[{label:"INV-1042",sub:"Acme Ltd · due 12 Feb 2026 · $4,210.00"}],ask:"List the invoices that are past due but still marked Due"},
    {id:"cancel_no_reason",title:"Cancellations with no reason recorded",severity:"low",status:"found",count:3,detail:"3 cancelled subscriptions have no cancellation reason.",why:"Churn analysis by reason is impossible.",fix:"Make the reason required and backfill recent ones.",examples:[],ask:"List cancelled subscriptions with no reason"},
    {id:"no_country",title:"Customers missing a billing country",severity:"low",status:"clear",count:0,detail:"Every customer has a billing country.",why:"",fix:"",examples:[]}],
  notes:["Sample data for preview. The data-hygiene skill replaces it with live results."]
};
const DATA = window.__SF_DATA__ || SAMPLE;

(function(){
  const $ = s => document.querySelector(s);
  const esc = s => String(s ?? "").replace(/[&<>"']/g, c => ({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c]));
  const fmtDate = d => d ? new Date(d+"T00:00:00").toLocaleDateString("en-GB",{day:"numeric",month:"short",year:"numeric"}) : "";
  const checks = DATA.checks || [];
  const found = checks.filter(c => c.status === "found");
  const n = sev => found.filter(c => c.severity === sev).length;
  $("#meta").textContent = `${DATA.account || ""}${DATA.account ? " · " : ""}${fmtDate(DATA.generated_at)} · ${checks.length} checks run`;
  $("#title").textContent = found.length ? `${found.length} issue${found.length===1?"":"s"} to look at` : "No problems found";
  $("#sum").innerHTML = [["high","high"],["medium","medium"],["low","low"]].filter(([s]) => n(s)).map(([s,c]) => `<span class="pill ${c}">${n(s)} ${s}</span>`).join("") +
    (DATA === SAMPLE ? ` <span class="badge">Sample data</span>` : "");

  let filter = found.length ? "found" : "all";
  const sevColor = {high:"var(--bad)",medium:"var(--warn)",low:"var(--muted)"};
  function render(){
    document.querySelectorAll(".chip").forEach(c => c.setAttribute("aria-pressed", c.dataset.f === filter));
    const rows = filter === "found" ? found : checks;
    $("#list").innerHTML = rows.map((c,i) => {
      const color = c.status === "clear" ? "var(--good)" : c.status === "unavailable" ? "var(--line)" : sevColor[c.severity] || "var(--muted)";
      const right = c.status === "unavailable" ? `<span class="pill unavailable">Not checked</span>` : c.status === "clear" ? `<span class="pill clear">Clear</span>` : `<span class="count">${(c.count ?? 0).toLocaleString()}</span>`;
      const ex = (c.examples||[]).length ? `<div class="ex">${c.examples.map(e => `<div><strong>${esc(e.label)}</strong><span class="muted">${esc(e.sub)}</span></div>`).join("")}</div>` : "";
      return `<details ${i===0 && c.status==="found" ? "open" : ""}><summary><span class="sev" style="background:${color}"></span><span><strong>${esc(c.title)}</strong><br><span class="small muted">${c.status === "found" ? esc(c.severity) + " priority" : c.status === "clear" ? "Nothing found" : esc(c.detail)}</span></span>${right}</summary>
        <div class="body"><p style="margin:0">${esc(c.detail)}</p>${c.why ? `<p class="small muted" style="margin:0"><strong>Why it matters:</strong> ${esc(c.why)}</p>` : ""}${c.fix ? `<p class="small muted" style="margin:0"><strong>How to fix:</strong> ${esc(c.fix)}</p>` : ""}${ex}${c.ask ? `<button class="btn" data-ask="${esc(c.ask)}">Show me these</button>` : ""}</div></details>`;
    }).join("") || `<p class="muted">No checks to show.</p>`;
    document.querySelectorAll("[data-ask]").forEach(b => b.onclick = () => {
      const s = b.dataset.ask;
      const t = $("#toast"); const show = m => { t.textContent = m; t.hidden = false; clearTimeout(show.t); show.t = setTimeout(() => t.hidden = true, 4000); };
      try { navigator.clipboard.writeText(s).then(() => show(`Copied "${s}". Paste it in the chat.`), () => show(`Say: "${s}"`)); } catch(e) { show(`Say: "${s}"`); }
    });
  }
  document.querySelectorAll(".chip").forEach(c => c.onclick = () => { filter = c.dataset.f; render(); });
  render();
  $("#notes").innerHTML = (DATA.notes||[]).map(x => `<li>${esc(x)}</li>`).join("");
})();
</script>
